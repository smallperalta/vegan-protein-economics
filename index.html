fetch('data/protein_foods.csv') .then(response => response.text()) .then(csv => { const rows = csv.split('\n').slice(1) // skip headers const x = [], y = [], labels = [], categories = [] rows.forEach(r => { if(!r.trim()) return const c = r.split(';') // Parse numbers safely const protein100 = parseFloat(c[5]) const pricePerProtein = parseFloat(c[6]) if(isNaN(protein100) || isNaN(pricePerProtein)) return x.push(protein100) y.push(pricePerProtein) labels.push(c[0]) categories.push(c[1]) }) console.log(x, y, labels, categories) // debug // Plotly trace Plotly.newPlot('chart', [{ x:x, y:y, text:labels, mode:'markers', type:'scatter' }], { xaxis:{title:"Protein (g per 100g)"}, yaxis:{title:"Price (€ per 100g protein)"} }) })
